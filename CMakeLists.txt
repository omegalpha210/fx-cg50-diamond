cmake_minimum_required(VERSION 3.15)
project(Diamond VERSION 0.1.0 LANGUAGES C ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(DG_HOST "Build host validation and renderer" OFF)
set(DG_ENGINE src/core/board.c src/core/engine.c src/game/game.c)
set(DG_STORAGE src/storage/codec.c src/storage/transaction.c src/storage/native.c)
set(DG_UI src/ui/app.c src/ui/entry.c src/ui/render.c src/ui/draw.c src/ui/menus.c src/ui/game_view.c)
if(DG_HOST)
 find_package(Python3 REQUIRED COMPONENTS Interpreter)
 add_compile_definitions(DG_HOST)
 add_compile_options(-Wall -Wextra -Werror -Wshadow -Wconversion -Wstrict-prototypes -UNDEBUG)
 option(DG_SANITIZE "Enable undefined behavior sanitizer" OFF)
 if(DG_SANITIZE)
  add_compile_options(-fsanitize=undefined -fno-sanitize-recover=undefined -g)
  add_link_options(-fsanitize=undefined)
 endif()
 add_library(diamond_core ${DG_ENGINE} src/ai/ai.c ${DG_STORAGE} src/platform/power.c)
 target_include_directories(diamond_core PUBLIC include)
 add_executable(test_engine tests/test_engine.c tests/test_ai.c)
 target_link_libraries(test_engine diamond_core)
 add_executable(test_storage tests/test_storage.c)
 target_link_libraries(test_storage diamond_core)
 add_executable(ai_audit tools/ai_audit.c)
 target_link_libraries(ai_audit diamond_core)
 if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tools/ai_strength.c")
  add_executable(ai_strength tools/ai_strength.c)
  target_link_libraries(ai_strength diamond_core)
 endif()
 if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tools/ai_fixtures.c")
  add_executable(ai_fixtures tools/ai_fixtures.c)
  target_link_libraries(ai_fixtures diamond_core)
 endif()
 enable_testing()
 add_test(NAME engine COMMAND test_engine)
 add_test(NAME storage COMMAND test_storage)
 add_test(NAME publication COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/tools/public_snapshot.py" --check)
 if(TARGET ai_fixtures)
  add_test(NAME ai_fixtures COMMAND ai_fixtures)
 endif()
 if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/ui/render.c" AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/ui/app.c" AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tools/capture.c" AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_ui.c")
  add_library(diamond_ui ${DG_UI})
  target_link_libraries(diamond_ui PUBLIC diamond_core)
  add_executable(test_difficulty tests/test_difficulty.c)
  target_link_libraries(test_difficulty diamond_ui)
  add_test(NAME difficulty COMMAND test_difficulty)
  set_tests_properties(difficulty PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
  add_executable(capture tools/capture.c)
  target_link_libraries(capture diamond_ui)
  add_executable(render_bench tools/render_bench.c)
  target_link_libraries(render_bench diamond_ui)
  add_executable(engine_golden tools/engine_golden.c)
  target_link_libraries(engine_golden diamond_core)
  add_test(NAME ui_freeze COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/tools/verify_ui_freeze.py" --executable $<TARGET_FILE:engine_golden> --ai-executable $<TARGET_FILE:ai_strength>)
  add_executable(test_ui tests/test_ui.c)
  target_link_libraries(test_ui diamond_ui)
  add_test(NAME ui COMMAND test_ui)
  add_executable(test_render_geometry tests/test_render_geometry.c)
  target_link_libraries(test_render_geometry diamond_ui)
  add_test(NAME render_geometry COMMAND test_render_geometry)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_app.c")
   add_executable(test_app tests/test_app.c)
   target_link_libraries(test_app diamond_ui)
   add_test(NAME app COMMAND test_app)
  endif()
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_native.c")
   add_executable(test_native tests/test_native.c)
   target_include_directories(test_native BEFORE PRIVATE tests/native)
   target_link_libraries(test_native diamond_ui)
   add_test(NAME native COMMAND test_native)
  endif()
 endif()
 add_executable(test_power tests/test_power.c)
 target_link_libraries(test_power diamond_core)
 add_test(NAME power COMMAND test_power)
else()
 include(GenerateG3A)
 find_package(Gint 2.11 REQUIRED)
 add_executable(diamond ${DG_ENGINE} src/ai/ai.c ${DG_STORAGE} ${DG_UI} src/platform/main.c src/platform/power.c src/platform/power_syscalls.S)
 target_include_directories(diamond PRIVATE include)
 target_compile_options(diamond PRIVATE -Os -Wall -Wextra -Werror -Wshadow -Wconversion -Wstrict-prototypes -Wframe-larger-than=2048 -fstack-usage)
 target_link_options(diamond PRIVATE "-Wl,-Map=${CMAKE_CURRENT_BINARY_DIR}/diamond.map")
 target_link_libraries(diamond Gint::Gint)
 set(DG_PACKAGE_DATE "2026.1001.0000" CACHE STRING "Deterministic G3A release date, yyyy.MMdd.hhmm")
 file(MAKE_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/dist")
 generate_g3a(TARGET diamond OUTPUT "${CMAKE_CURRENT_SOURCE_DIR}/dist/DIAMOND.g3a" NAME "DIAMOND" VERSION "00.01.0000" ICONS assets/icon-uns.png assets/icon-sel.png)
 add_custom_command(TARGET diamond POST_BUILD COMMAND fxgxa -e "${CMAKE_CURRENT_SOURCE_DIR}/dist/DIAMOND.g3a" --internal=@DIAMOND "--date=${DG_PACKAGE_DATE}" COMMAND fxgxa -r "${CMAKE_CURRENT_SOURCE_DIR}/dist/DIAMOND.g3a" VERBATIM)
endif()
